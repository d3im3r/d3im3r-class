    \begin{center}
        \IfFileExists{#2}{
            \includegraphics[
            width=#1,
            keepaspectratio
            ]{#2}
        }{
            \fbox{
                \parbox[c][1.5cm][c]{0.75\linewidth}{
                    \centering
                    \scriptsize
                    \color{ExamGray}
                    Imagen no encontrada\\[1mm]
                    \texttt{\detokenize{#2}}
                }
            }
        }
    \end{center}
}

\newenvironment{answers}{
    \begin{adjustwidth}{\AnswerIndent}{0cm}
        \begin{enumerate}[
            label=\textbf{\Alph*.},
            leftmargin=5mm,
            labelsep=1.5mm,
            itemsep=\AnswerItemSep,
            topsep=\AnswerTopSep,
            parsep=0pt,
            partopsep=0pt
            ]
        }{
        \end{enumerate}
    \end{adjustwidth}
}

% =========================================================
% RESPUESTAS EN UNA LÍNEA
% =========================================================

\newenvironment{inlineanswers}{
    \begin{adjustwidth}{\AnswerIndent}{0cm}
        \begin{enumerate*}[
            label=\textbf{\Alph*.},
            itemjoin=\hspace{\InlineAnswerSep},
            itemjoin*=\hspace{\InlineAnswerSep},
            itemsep=0pt
            ]
        }{
        \end{enumerate*}
    \end{adjustwidth}
}


% =========================================================
% RESPUESTAS EN CUADRÍCULA 2 x 2
% =========================================================

\newenvironment{gridanswers}{
    \begin{adjustwidth}{\AnswerIndent}{0cm}
        \begin{multicols}{2}
            \begin{enumerate}[
                label=\textbf{\Alph*.},
                leftmargin=5mm,
                labelsep=1.5mm,
                itemsep=\GridAnswerSep,
                topsep=\AnswerTopSep,
                parsep=0pt,
                partopsep=0pt
                ]
            }{
            \end{enumerate}
        \end{multicols}
    \end{adjustwidth}
}
