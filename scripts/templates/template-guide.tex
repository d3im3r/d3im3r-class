\newcommand{\smartimage}[2][0.9\linewidth]{%
    \ifnum\UseImages=1
    \IfFileExists{#2}{%
        \begin{center}
            \includegraphics[width=#1]{#2}
        \end{center}
    }{%
        \begin{center}
            \fbox{%
                \parbox{#1}{%
                    \centering
                    \scriptsize
                    Imagen no encontrada:\\
                    \texttt{#2}
                }
            }
        \end{center}
    }%
    \fi
}

\newcommand{\smartlogo}[2][1.2cm]{%
    \ifnum\UseImages=1
    \IfFileExists{#2}{%
        \includegraphics[height=#1]{#2}
    }{%
        \fbox{%
            \parbox[c][#1][c]{2cm}{%
                \centering
                \scriptsize
                Logo no encontrado
            }
        }
    }%
    \fi
}
